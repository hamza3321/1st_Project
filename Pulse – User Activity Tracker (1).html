Add it to your app

Copy the tracker script, then one call:

<script src="tracker.js"></script>
<script>
  Pulse.init({
    endpoint: "/api/track" // optional
  });
</script>

// Read data any time
Pulse.getData();
// Custom event
Pulse.track("signup", { plan: "team" });
